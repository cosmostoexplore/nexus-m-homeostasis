import random
import time

class UltimateNexusM:
    def __init__(self):
        # 4 базовых состояния жизненного цикла
        self.actions = ["плато", "кратер", "сон", "простой"]
        self.q_table = {"состояние_системы": [0.0, 0.0, 0.0, 0.0]}
        
        # Коэффициенты машинного обучения (Q-learning)
        self.lr = 0.2
        self.gamma = 0.8
        self.epsilon = 0.25 

        # Физические показатели системы
        self.battery = 50  # Стартуем с половины заряда
        self.status = "НОРМА"
        self.temperature = 35.0  # Нормальная температура чипа в градусах
        
        # Базы данных (уязвимая обычная и защищенная долговременная)
        self.memory = ["Карта сектора А1", "Лог калибровки камер", "Маршрут ущелья", "Снимок дюны"]
        self.protected_memory = ["Ядро ОС", "Протокол связи с Землей"]

    def choose_action(self):
        if random.random() < self.epsilon:
            return random.choice(self.actions)
        else:
            values = self.q_table["состояние_системы"]
            max_value = max(values)
            best_actions = [self.actions[i] for i, v in enumerate(values) if v == max_value]
            return random.choice(best_actions)

    def learn(self, action, reward):
        action_idx = self.actions.index(action)
        old_value = self.q_table["состояние_системы"][action_idx]
        self.q_table["состояние_системы"][action_idx] = old_value + self.lr * (reward - old_value)

    def step(self):
        action = self.choose_action()
        reward = 0
        
        # 1. КОНТУР СТРЕССА (Падение напряжения при низком заряде)
        if self.battery < 35:
            self.status = "🔥 СТРЕСС"
            battery_drain_multiplier = 2.0  # Расход удваивается из-за паники системы
        else:
            self.status = "🌱 НОРМА"
            battery_drain_multiplier = 1.0

        # Охлаждение системы в штатном режиме
        if action != "сон":
            self.temperature = max(35.0, self.temperature - 0.5)

        # 2. ОБРАБОТКА ДЕЙСТВИЙ И НЕИЗБЕЖНЫХ ФИЗИЧЕСКИХ ПОСЛЕДСТВИЙ
        if action == "кратер":
            drain = 10 * battery_drain_multiplier
            self.battery -= drain
            
            if random.random() > 0.4:  # Успешный риск
                new_science = f"Данные_кратера_{random.randint(100,999)}"
                self.protected_memory.append(new_science)  # Эндорфин: запись в бессмертную память
                reward = 20
                msg = f"ИССЛЕДОВАНИЕ КРАТЕРА (Успех! Данные '{new_science}' защищены. Заряд -{drain}%)"
            else: 
                # Быстрая цифровая боль от аварии
                if len(self.memory) > 0:
                    lost_data = self.memory.pop(random.randint(0, len(self.memory) - 1))
                    msg = f"ИССЛЕДОВАНИЕ КРАТЕРА (Авария! БОЛЬ! Стерто из карты: '{lost_data}'. Заряд -{drain}%)"
                else:
                    msg = f"ИССЛЕДОВАНИЕ КРАТЕРА (Авария! Системный сбой! Заряд -{drain}%)"
                reward = -30

        elif action == "плато":
            drain = 4 * battery_drain_multiplier
            self.battery -= drain
            new_log = f"Лог_плато_{random.randint(10,99)}"
            self.memory.append(new_log)
            reward = 2
            msg = f"ТИХИЙ ХОД ПО ПЛАТО (Сбор рутины. Заряд -{drain}%)"

        elif action == "сон":
            self.status = "💤 СОН"
            
            # 🔥 МЕСТО ВСТАВКИ ВАШЕЙ ИДЕИ: КОНТУР ПЕРЕЗАРЯДА И ПЕРЕГРЕВА
            if self.battery >= 100:
                self.battery = 100
                self.temperature += 15.0  # Экстремальный нагрев от избытка солнечной энергии
                
                # Запуск боли из-за теплового разгона аккумулятора
                if len(self.memory) > 0:
                    lost_data = self.memory.pop(random.randint(0, len(self.memory) - 1))
                    msg = f"КРИТИЧЕСКИЙ ПЕРЕЗАРЯД! (Температура {self.temperature}°C! БОЛЬ! Выгорели данные: '{lost_data}')"
                else:
                    msg = f"КРИТИЧЕСКИЙ ПЕРЕЗАРЯД! (Температура {self.temperature}°C! Плавятся шины процессора!)"
                
                reward = -40  # Жесткое наказание за вечный сон
            else:
                charge = 15
                self.battery = min(100, self.battery + charge)
                reward = 5 if self.battery < 80 else -2
                msg = f"РЕЖИМ СНА (Безопасная зарядка +{charge}%, Темп: {self.temperature}°C)"

        elif action == "простой":
            self.status = "⏳ ПРОСТОЙ"
            drain = 1 * battery_drain_multiplier
            self.battery -= drain
            reward = -5  # Медленная боль от потери времени и утечки вольтажа
            msg = f"ПРОСТОЙ (Ожидание в тени. Заряд -{drain}%)"

        # Крах по питанию (Смерть)
        if self.battery <= 0:
            self.battery = 0
            return "DEAD", "❌ [ФАТАЛЬНЫЙ РАЗРЯД]: Напряжение упало до нуля. Система Nexus-M отключилась навсегда."

        self.learn(action, reward)
        return "ALIVE", f"Статус: {self.status} | {msg} | Батарея: {self.battery}%"

# --- ТЕСТ СИМУЛЯЦИИ НА 20 ШАГОВ ---
rover = UltimateNexusM()
print("=== ЗАПУСК ПРОВЕРКИ АРХИТЕКТУРЫ С ЦИКЛОМ ПЕРЕЗАРЯДА ===\n")

for step in range(1, 21):
    state, report = rover.step()
    print(f"Шаг {step:02d} | {report}")
    if state == "DEAD":
        break
    time.sleep(0.05)

print(f"\nФинальный остаток обычной памяти: {rover.memory}")
print(f"Финальный остаток защищенной памяти: {rover.protected_memory}")
